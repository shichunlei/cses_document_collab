use yrs::undo::UndoManager;
use yrs::{Doc,Options,OffsetKind,Text,ReadTxn,Transact,GetString,StateVector};
fn manager(doc:&Doc,text:&yrs::TextRef,prior:Option<&UndoManager<()>>)->UndoManager<()> {let mut m=UndoManager::with_options(yrs::undo::Options{capture_timeout_millis:0,init_undo_stack:prior.map(|m|m.undo_stack().to_vec()).unwrap_or_default(),init_redo_stack:prior.map(|m|m.redo_stack().to_vec()).unwrap_or_default(),..Default::default()});m.expand_scope(doc,text);m}
fn main(){let doc=Doc::with_options(Options{skip_gc:true,offset_kind:OffsetKind::Utf16,..Default::default()});let text=doc.get_or_insert_text("body");let mut m=manager(&doc,&text,None);let mut rnd=17u64;for step in 0..2500{rnd=rnd.wrapping_mul(6364136223846793005).wrapping_add(1);let fork=doc.fork_for_history().unwrap();let ft=fork.get_or_insert_text("body");let mut fm=manager(&fork,&ft,Some(&m));let len=text.len(&doc.transact());let pos=if len==0 {0}else{(rnd>>8) as u32 %(len+1)};match rnd%5{0|1=>{let s=if rnd%2==0{"A"}else{"b"};text.insert(&mut doc.transact_mut(),pos,s);ft.insert(&mut fork.transact_mut(),pos,s);},2 if len>0=>{let pos=pos.min(len-1);text.remove_range(&mut doc.transact_mut(),pos,1);ft.remove_range(&mut fork.transact_mut(),pos,1);},3=>{assert_eq!(m.undo_blocking(),fm.undo_blocking(),"step {step}");},4=>{assert_eq!(m.redo_blocking(),fm.redo_blocking(),"step {step}");},_=>()};assert_eq!(text.get_string(&doc.transact()),ft.get_string(&fork.transact()),"step {step}");assert_eq!(doc.transact().encode_state_as_update_v1(&StateVector::default()),fork.transact().encode_state_as_update_v1(&StateVector::default()),"wire step {step}");}println!("2500 differential text edit/undo/redo forks passed");}
